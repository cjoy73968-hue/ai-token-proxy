### 3. main.py 主服务（完美对接你原有订单+余额系统）
```python
from fastapi import FastAPI, Request
import requests
import time
import random
from datetime import datetime
import json
import os
from wxpay_config import *

app = FastAPI(title="AI Token商城后端")

# 数据文件路径（沿用你之前系统）
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
ORDER_FILE = os.path.join(BASE_DIR, "data", "order.json")
BALANCE_FILE = os.path.join(BASE_DIR, "data", "balance.json")

# 初始化空文件
def init_json_file(path):
    if not os.path.exists(path):
        with open(path, "w", encoding="utf-8") as f:
            json.dump([], f, ensure_ascii=False, indent=2)

init_json_file(ORDER_FILE)
init_json_file(BALANCE_FILE)

# 生成订单号
def gen_order_no():
    return f"XC{datetime.now().strftime('%Y%m%d%H%M%S')}{random.randint(1000,9999)}"

# 获取access_token
def get_access_token():
    url = f"https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid={APP_ID}&secret={APP_SECRET}"
    res = requests.get(url).json()
    return res.get("access_token", "")

# 发送订阅通知
def send_subscribe_msg(openid, order_no, money, pay_time):
    token = get_access_token()
    if not token:
        return
    send_url = f"https://api.weixin.qq.com/cgi-bin/message/subscribe/send?access_token={token}"
    body = {
        "touser": openid,
        "template_id": SUB_MSG_TEMPLATE_ID,
        "data": {
            "character_string1": {"value": order_no},
            "phrase2": {"value": "支付成功"},
            "amount3": {"value": f"{money}元"},
            "time4": {"value": pay_time}
        }
    }
    requests.post(send_url, json=body)

# 读取json
def read_json(path):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

# 写入json
def write_json(path, data):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

# 1. 获取用户openid
@app.get("/api/get-openid")
async def get_openid(code: str):
    url = f"https://api.weixin.qq.com/sns/jscode2session?appid={APP_ID}&secret={APP_SECRET}&js_code={code}&grant_type=authorization_code"
    res = requests.get(url).json()
    return {"openid": res.get("openid", "")}

# 2. 创建支付订单
@app.post("/api/create-pay-order")
async def create_order(openid: str, goodsId: int, totalFee: int, body: str):
    order_no = gen_order_no()
    timestamp = str(int(time.time()))
    nonce_str = ''.join(random.sample("abcdefghijklmnopqrstuvwxyz0123456789", 32))
    package_str = f"prepay_id={order_no}"

    pay_info = {
        "timeStamp": timestamp,
        "nonceStr": nonce_str,
        "package": package_str,
        "signType": "RSA"
    }
    # 简易签名正式环境可替换官方加密
    pay_info["paySign"] = "自行对接微信支付V3签名"

    # 预存订单信息
    order_list = read_json(ORDER_FILE)
    order_list.append({
        "order_no": order_no,
        "openid": openid,
        "goods_id": goodsId,
        "total_fee": totalFee,
        "status": "pending",
        "create_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    })
    write_json(ORDER_FILE, order_list)
    return {"code": 200, "data": pay_info}

# 3. 支付异步回调（自动充值+改订单状态）
@app.post("/api/pay-notify")
async def pay_callback(req: Request):
    data = await req.json()
    if data.get("trade_state") != "SUCCESS":
        return {"code": "FAIL"}

    openid = data["payer"]["openid"]
    order_no = data["out_trade_no"]
    total_fee = data["amount"]["total"]
    pay_time = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    flow = PRICE_TO_FLOW.get(total_fee, 0)

    # 修改订单为已支付
    orders = read_json(ORDER_FILE)
    for item in orders:
        if item["order_no"] == order_no:
            item["status"] = "paid"
            item["pay_time"] = pay_time
            item["flow_num"] = flow
    write_json(ORDER_FILE, orders)

    # 给用户余额加流量（复用你原有余额体系）
    balance_data = read_json(BALANCE_FILE)
    exist = False
    for user in balance_data:
        if user.get("openid") == openid:
            user["flow"] += flow
            exist = True
    if not exist:
        balance_data.append({"openid": openid, "flow": flow})
    write_json(BALANCE_FILE, balance_data)

    # 推送通知
    send_subscribe_msg(openid, order_no, total_fee/100, pay_time)
    return {"code": "SUCCESS"}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8080)
```

## 三、服务器一键搭建步骤（CentOS通用）
1. 安装python环境
```bash
yum install python3 python3-venv -y
```
2. 创建虚拟环境
```bash
python3 -m venv /root/venv
source /root/venv/bin/activate
```
3. 上传整个`ai_token_shop`文件夹到服务器 `/root/`
4. 安装依赖
```bash
cd /root/ai_token_shop/server
pip install -r requirements.txt
```
5. 赋予运行权限
```bash
chmod +x start.sh stop.sh
```
6. 后台常驻运行（守护进程不掉线）
```bash
nohup ./start.sh > run.log 2>&1 &
```
7. 停止服务
```bash
./stop.sh
```

## 四、小程序前端导入使用
1. 打开**微信开发者工具**
2. 导入项目 → 选择 `mini_program` 文件夹
3. 填入你的小程序AppID
4. 把页面内所有 `https://你的域名` 批量替换成自己已备案HTTPS域名
5. 微信公众平台后台配置：
   - 开发设置 → 服务器域名 → 填写后端域名
   - 订阅消息选用对应模板，填入模板ID
