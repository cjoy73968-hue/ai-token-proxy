## 二、整体流程（一句话）
用户点击购买 → 前端请求后端下单 → 后端调用微信支付**统一下单** → 返回支付参数给前端 → 前端**调起微信支付** → 支付成功微信**异步回调**你的服务器 → 服务器**发订阅消息**给用户。

### 2. 支付逻辑 + 订阅授权（.js）
```javascript
Page({
  data: {
    isLoading: false,
    orderId: 'ORDER_20260519_001', // 后端生成
    price: 990, // 单位：分（9.9元）
    openid: ''  // 需提前通过 wx.login 获取
  },

  onLoad() {
    this.getUserOpenId(); // 先拿openid
  },

  // 1. 获取用户openid（必须）
  getUserOpenId() {
    let that = this;
    wx.login({
      success(res) {
        if (res.code) {
          // 传给后端换取 openid
          wx.request({
            url: 'https://你的域名/api/getOpenid',
            data: { code: res.code },
            success(r) {
              that.setData({ openid: r.data.openid });
            }
          });
        }
      }
    });
  },

  // 2. 点击支付按钮
  async payOrder() {
    let that = this;
    if (that.data.isLoading) return;
    that.setData({ isLoading: true });

    // 🔔 关键：支付前先请求订阅消息授权（必须用户点击触发）
    const templateId = '你的支付成功通知模板ID';
    wx.requestSubscribeMessage({
      tmplIds: [templateId],
      success(res) {
        // 用户允许后，再发起支付
        if (res[templateId] === 'accept') {
          that.createOrderAndPay();
        } else {
          wx.showToast({ title: '你拒绝了通知', icon: 'none' });
          that.setData({ isLoading: false });
        }
      },
      fail() {
        wx.showToast({ title: '授权失败', icon: 'none' });
        that.setData({ isLoading: false });
      }
    });
  },

  // 3. 后端下单 + 前端调起支付
  createOrderAndPay() {
    let that = this;
    // 向后端请求预支付订单
    wx.request({
      url: 'https://你的域名/api/unifiedOrder',
      method: 'POST',
      data: {
        openid: that.data.openid,
        orderId: that.data.orderId,
        totalFee: that.data.price,
        body: 'AI算力套餐-9.9元体验装'
      },
      success(res) {
        if (res.data.success) {
          // 调起微信支付
          wx.requestPayment({
            ...res.data.payInfo, // 后端返回的支付参数
            success(payRes) {
              // 前端支付成功提示
              wx.showToast({ title: '支付成功', icon: 'success' });
              // 实际发消息靠后端回调，这里不用发
            },
            fail() {
              wx.showToast({ title: '支付取消/失败', icon: 'none' });
            },
            complete() {
              that.setData({ isLoading: false });
            }
          });
        }
      },
      fail() {
        wx.showToast({ title: '下单失败', icon: 'none' });
        that.setData({ isLoading: false });
      }
    });
  }
});
```

## 四、后端核心代码（Node.js示例，你可转Java/PHP/Python）

### 1. 统一下单（获取prepay_id）
```javascript
// 后端 /api/unifiedOrder
const axios = require('axios');
const crypto = require('crypto');

// 配置
const config = {
  appid: '小程序AppID',
  mchid: '商户号',
  apiV3Key: 'APIv3密钥',
  serialNo: '商户证书序列号',
  privateKey: '商户私钥文本' // 从证书pem文件读取
};

// 生成签名（简化）
function generateSignature(method, url, timestamp, nonceStr, body) {
  let signatureStr = `${method}\n${url}\n${timestamp}\n${nonceStr}\n${body}\n`;
  let sign = crypto.sign('sha256', Buffer.from(signatureStr), config.privateKey);
  return sign.toString('base64');
}

// 统一下单
async function unifiedOrder(openid, orderId, totalFee, body) {
  const timestamp = Math.floor(Date.now() / 1000).toString();
  const nonceStr = crypto.randomBytes(16).toString('hex');
  const reqBody = JSON.stringify({
    appid: config.appid,
    mchid: config.mchid,
    description: body,
    out_trade_no: orderId,
    amount: { total: totalFee, currency: 'CNY' },
    payer: { openid },
    notify_url: 'https://你的域名/api/payNotify' // 支付结果回调
  });

  const url = '/v3/pay/transactions/jsapi';
  const signature = generateSignature('POST', url, timestamp, nonceStr, reqBody);
  const token = `WECHATPAY2-SHA256-RSA2048 mchid="${config.mchid}",nonce_str="${nonceStr}",signature="${signature}",timestamp="${timestamp}",serial_no="${config.serialNo}"`;

  const res = await axios.post(`https://api.mch.weixin.qq.com${url}`, reqBody, {
    headers: {
      'Authorization': token,
      'Content-Type': 'application/json'
    }
  });

  // 生成前端支付参数
  const prepayId = res.data.prepay_id;
  const payInfo = {
    appId: config.appid,
    timeStamp: timestamp,
    nonceStr: nonceStr,
    package: `prepay_id=${prepayId}`,
    signType: 'RSA'
  };
  // 对payInfo签名
  const paySignStr = `${config.appid}\n${timestamp}\n${nonceStr}\n${payInfo.package}\n`;
  payInfo.paySign = crypto.sign('sha256', Buffer.from(paySignStr), config.privateKey).toString('base64');

  return { success: true, payInfo };
}

module.exports = { unifiedOrder };
```

### 2. 支付结果回调（关键：发订阅消息）
```javascript
// 后端 /api/payNotify（微信异步POST）
const axios = require('axios');

// 1. 验证回调签名（略，必须做）
// 2. 解析支付结果
async function payNotify(req, res) {
  const notifyData = req.body;
  const tradeState = notifyData.trade_state; // SUCCESS：支付成功

  if (tradeState === 'SUCCESS') {
    const openid = notifyData.payer.openid;
    const orderId = notifyData.out_trade_no;
    const totalFee = notifyData.amount.total / 100;
    const successTime = notifyData.success_time;

    // 🔔 关键：支付成功 → 发订阅消息
    await sendSubscribeMessage(openid, orderId, totalFee, successTime);

    // 更新订单状态
    // db.updateOrder({ orderId, status: 'PAID' });
  }

  // 回复微信
  res.json({ code: 'SUCCESS', message: '成功' });
}

// 3. 发送订阅消息（支付成功通知）
async function sendSubscribeMessage(openid, orderId, totalFee, successTime) {
  // 先获取 access_token（2小时有效，缓存）
  const tokenRes = await axios.get(`https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid=${config.appid}&secret=${config.appSecret}`);
  const accessToken = tokenRes.data.access_token;

  // 订阅消息模板（按你选的模板字段）
  const templateId = '你的支付成功模板ID';
  const message = {
    touser: openid,
    template_id: templateId,
    page: 'pages/order/order', // 点击跳转页面
    data: {
      // 字段名必须和模板完全一致
      character_string1: { value: orderId },        // 订单编号
      phrase2: { value: '支付成功' },               // 订单状态
      amount3: { value: `${totalFee}元` },          // 支付金额
      time4: { value: successTime }                // 支付时间
    }
  };

  // 调用微信接口发消息
  await axios.post(`https://api.weixin.qq.com/cgi-bin/message/subscribe/send?access_token=${accessToken}`, message);
}

module.exports = { payNotify };
```

## 八、避坑要点（必看）
1. **订阅授权必须在用户点击（payOrder）里调用**，不能onLoad自动弹
2. **发消息必须在后端回调**，不要前端发（不安全、易丢失）
3. **金额单位是分**（9.9元 = 990）
4. **access_token 必须缓存（2小时）**，不要每次重新获取
5. **回调接口必须验证签名**，防止伪造通知
