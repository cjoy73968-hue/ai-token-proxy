## 一、完整 main.py 代码
```python
from fastapi import FastAPI, Request, HTTPException
import httpx
import time
from datetime import datetime

app = FastAPI(title="AI Token代理服务")

# ===================== 全局配置区 自行修改 =====================
# 上游模型配置
UPSTREAM_API_KEY = "你的上游官方API密钥"
UPSTREAM_API_URL = "https://api.deepseek.com/v1/chat/completions"
TIMEOUT_SECONDS = 120

# 限流配置 单IP每分钟最大请求数
LIMIT_PER_MIN = 80
ip_request_record = {}

# 流量计价：单次请求平均消耗万Token
AVG_CONSUME_WAN_TOKEN = 0.05

# 1. 用户密钥 & 剩余流量 字典 (单位：万Token)
USER_BALANCE = {
    "tk202605180001": 1150,
    "tk202605180002": 6000,
    "tk202605180003": 23000
}

# 2. 支付订单管理库 简易内存版(重启清空，正式用Excel/Json持久化)
# 字段：订单号,微信昵称,金额,套餐,绑定密钥,状态:待付款/已付款/已发货/已过期
PAY_ORDER_LIST = []
# 套餐映射 {金额:初始万Token}
PRICE_PACKAGE = {
    9.9: 1150,
    49: 6000,
    188: 23000
}
# =================================================================

# IP限流中间件
@app.middleware("http")
async def ip_limit_middleware(request: Request, call_next):
    client_ip = request.client.host
    now_ts = int(time.time())
    if client_ip not in ip_request_record or now_ts - ip_request_record[client_ip]["ts"] > 60:
        ip_request_record[client_ip] = {"cnt": 1, "ts": now_ts}
    else:
        ip_request_record[client_ip]["cnt"] += 1
        if ip_request_record[client_ip]["cnt"] > LIMIT_PER_MIN:
            raise HTTPException(status_code=429, detail="请求过于频繁，请稍后重试")
    return await call_next(request)

# 核心对话转发接口
@app.post("/v1/chat/completions")
async def chat_proxy(request: Request):
    # 1. 提取用户密钥
    auth_header = request.headers.get("Authorization", "")
    if not auth_header.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="授权格式错误")
    user_key = auth_header.replace("Bearer ", "").strip()

    # 2. 校验密钥是否存在
    if user_key not in USER_BALANCE:
        raise HTTPException(status_code=401, detail="无效代理密钥")
    
    # 3. 判断余额是否充足
    if USER_BALANCE[user_key] <= 0:
        raise HTTPException(status_code=403, detail="流量已耗尽，请续费后使用")
    
    # 4. 获取请求体 不落地存储
    req_body = await request.json()

    # 5. 转发上游接口
    headers = {
        "Authorization": f"Bearer {UPSTREAM_API_KEY}",
        "Content-Type": "application/json"
    }
    try:
        async with httpx.AsyncClient(timeout=TIMEOUT_SECONDS) as client:
            resp = await client.post(UPSTREAM_API_URL, json=req_body, headers=headers)
    except Exception as e:
        raise HTTPException(status_code=500, detail="上游接口请求异常")

    # 6. 自动扣减流量
    USER_BALANCE[user_key] -= AVG_CONSUME_WAN_TOKEN

    return resp.json()

# 接口1：查询自身余额（客户自助查）
@app.get("/user/balance")
async def query_balance(token_key: str):
    if token_key not in USER_BALANCE:
        return {"code":401,"msg":"密钥不存在","balance":0}
    remain = round(USER_BALANCE[token_key],2)
    warn_tip = "流量充足"
    if remain < 100:
        warn_tip = "【预警】流量即将不足，请尽快续费"
    return {
        "code":200,
        "user_key":token_key,
        "remain_wan_token":remain,
        "tip":warn_tip
    }

# 接口2：商家创建支付订单
@app.post("/admin/create-order")
async def create_order(nickname: str, money: float, bind_key: str):
    if money not in PRICE_PACKAGE:
        return {"code":400,"msg":"无此套餐"}
    import uuid
    order_no = f"ORD{datetime.now().strftime('%Y%m%d%H%M%S')}{uuid.uuid4().hex[:6]}"
    init_flow = PRICE_PACKAGE[money]
    new_order = {
        "order_no":order_no,
        "nickname":nickname,
        "money":money,
        "init_flow":init_flow,
        "bind_key":bind_key,
        "status":"待付款",
        "create_time":datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    }
    PAY_ORDER_LIST.append(new_order)
    return {"code":200,"order_info":new_order}

# 接口3：商家确认付款、发放流量
@app.post("/admin/pay-success")
async def pay_success(order_no: str):
    for item in PAY_ORDER_LIST:
        if item["order_no"] == order_no and item["status"] == "待付款":
            item["status"] = "已付款已发货"
            # 给对应密钥充值流量
            target_key = item["bind_key"]
            add_flow = item["init_flow"]
            if target_key in USER_BALANCE:
                USER_BALANCE[target_key] += add_flow
            else:
                USER_BALANCE[target_key] = add_flow
            return {"code":200,"msg":"充值成功","order":item}
    return {"code":400,"msg":"订单不存在或已处理"}

# 接口4：商家查看全部订单
@app.get("/admin/all-order")
async def get_all_order():
    return {"code":200,"data":PAY_ORDER_LIST}

# 健康检测
@app.get("/")
async def health_check():
    return {"status":"running","note":"纯中转无数据留存，带余额管控+订单系统"}
```

## 三、三大核心功能使用说明
### 1. 自动流量管控
1. 用户每次调用自动扣 `0.05万Token`
2. 余额 ≤0 直接弹窗禁止使用，自动回收权限
3. 余额＜100万Token 自助查询显示**预警提示**

### 2. 客户自助查余额
访问地址：
```
https://你的域名/user/balance?token_key=客户密钥
```
手机浏览器直接打开就能看剩余流量

### 3. 内置简易支付订单管理（商家后台）
#### ① 创建订单
请求示例：
```
POST https://你的域名/admin/create-order?nickname=客户昵称&money=9.9&bind_key=tk202605180001
```
自动生成唯一订单号，录入待付款列表

#### ② 客户转账后确认发货充值
```
POST https://你的域名/admin/pay-success?order_no=生成的订单号
```
执行后：
- 订单状态变更
- 自动给对应密钥加上套餐流量
- 立刻恢复使用

#### ③ 查看全部订单记录
```
GET https://你的域名/admin/all-order
```

## 四、落地运营管理搭配方案
1. **密钥库存**
   Excel批量生成 `tk+日期+随机数` 密钥，提前填入 `USER_BALANCE` 赋初始0流量

2. **日常出单流程**
   1. 客户下单选套餐
   2. 调用创建订单接口生成单号
   3. 客户微信转账
   4. 后台确认付款 → 自动充值流量
   5. 自动发：密钥+调用地址+查余额链接+教程

3. **余额提醒**
   每天导出后台订单+密钥余额，筛选低于100万流量用户统一发微信预警文案

4. **闲置密钥回收**
   长期无消费、余额为0密钥，直接从 `USER_BALANCE` 删除，重新分配新客户
