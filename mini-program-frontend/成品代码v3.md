## 一、最终完整版：main.py（JSON持久化+余额+订单+支付）
```python
from fastapi import FastAPI, Request, HTTPException
import httpx
import time
import json
import os
import threading
from datetime import datetime
from pathlib import Path

app = FastAPI(title="AI Token代理服务（持久版）")

# ===================== 全局配置区 =====================
UPSTREAM_API_KEY = "你的上游API密钥"
UPSTREAM_API_URL = "https://api.deepseek.com/v1/chat/completions"
TIMEOUT_SECONDS = 120
LIMIT_PER_MIN = 80
AVG_CONSUME_WAN_TOKEN = 0.05  # 单次扣量（万Token）

# JSON数据文件路径
DATA_DIR = Path("./data")
BALANCE_FILE = DATA_DIR / "balance.json"
ORDER_FILE = DATA_DIR / "order.json"

# 线程锁（防止并发写坏文件）
lock = threading.Lock()

# 套餐配置
PRICE_PACKAGE = {
    9.9: 1150,
    49: 6000,
    188: 23000
}

# IP限流记录
ip_request_record = {}

# ===================== 工具函数：JSON持久化 =====================
def init_data_files():
    """初始化数据目录与文件"""
    DATA_DIR.mkdir(exist_ok=True)
    # 余额文件
    if not BALANCE_FILE.exists():
        with open(BALANCE_FILE, "w", encoding="utf-8") as f:
            json.dump({}, f, ensure_ascii=False, indent=2)
    # 订单文件
    if not ORDER_FILE.exists():
        with open(ORDER_FILE, "w", encoding="utf-8") as f:
            json.dump([], f, ensure_ascii=False, indent=2)

def load_balance() -> dict:
    """加载余额"""
    with lock:
        with open(BALANCE_FILE, "r", encoding="utf-8") as f:
            return json.load(f)

def save_balance(data: dict):
    """保存余额"""
    with lock:
        with open(BALANCE_FILE, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

def load_orders() -> list:
    """加载订单"""
    with lock:
        with open(ORDER_FILE, "r", encoding="utf-8") as f:
            return json.load(f)

def save_orders(data: list):
    """保存订单"""
    with lock:
        with open(ORDER_FILE, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

# 启动时初始化文件
init_data_files()
USER_BALANCE = load_balance()
PAY_ORDER_LIST = load_orders()

# ===================== 中间件：IP限流 =====================
@app.middleware("http")
async def ip_limit_middleware(request: Request, call_next):
    client_ip = request.client.host
    now_ts = int(time.time())
    if client_ip not in ip_request_record or now_ts - ip_request_record[client_ip]["ts"] > 60:
        ip_request_record[client_ip] = {"cnt": 1, "ts": now_ts}
    else:
        ip_request_record[client_ip]["cnt"] += 1
        if ip_request_record[client_ip]["cnt"] > LIMIT_PER_MIN:
            raise HTTPException(status_code=429, detail="请求过于频繁")
    return await call_next(request)

# ===================== 核心接口：对话转发 =====================
@app.post("/v1/chat/completions")
async def chat_proxy(request: Request):
    # 1. 提取密钥
    auth_header = request.headers.get("Authorization", "")
    if not auth_header.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="授权格式错误")
    user_key = auth_header.replace("Bearer ", "").strip()

    # 2. 校验密钥
    if user_key not in USER_BALANCE:
        raise HTTPException(status_code=401, detail="无效密钥")

    # 3. 余额判断
    if USER_BALANCE[user_key] <= 0:
        raise HTTPException(status_code=403, detail="流量已耗尽，请续费")

    # 4. 读取请求
    req_body = await request.json()

    # 5. 转发上游
    headers = {
        "Authorization": f"Bearer {UPSTREAM_API_KEY}",
        "Content-Type": "application/json"
    }
    try:
        async with httpx.AsyncClient(timeout=TIMEOUT_SECONDS) as client:
            resp = await client.post(UPSTREAM_API_URL, json=req_body, headers=headers)
            resp.raise_for_status()
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"上游错误：{str(e)}")

    # 6. 扣量 + 保存到文件
    USER_BALANCE[user_key] -= AVG_CONSUME_WAN_TOKEN
    save_balance(USER_BALANCE)

    return resp.json()

# ===================== 用户接口：查询余额 =====================
@app.get("/user/balance")
async def query_balance(token_key: str):
    if token_key not in USER_BALANCE:
        return {"code": 401, "msg": "密钥不存在", "balance": 0}
    remain = round(USER_BALANCE[token_key], 2)
    tip = "流量充足" if remain >= 100 else "【预警】流量不足，请尽快续费"
    return {
        "code": 200,
        "user_key": token_key,
        "remain_wan_token": remain,
        "tip": tip
    }

# ===================== 管理员接口：订单管理 =====================
@app.post("/admin/create-order")
async def create_order(nickname: str, money: float, bind_key: str):
    if money not in PRICE_PACKAGE:
        return {"code": 400, "msg": "无效套餐"}
    # 生成订单号
    order_no = f"ORD{datetime.now().strftime('%Y%m%d%H%M%S')}"
    init_flow = PRICE_PACKAGE[money]
    new_order = {
        "order_no": order_no,
        "nickname": nickname,
        "money": money,
        "init_flow": init_flow,
        "bind_key": bind_key,
        "status": "待付款",
        "create_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    }
    PAY_ORDER_LIST.append(new_order)
    save_orders(PAY_ORDER_LIST)
    return {"code": 200, "order": new_order}

@app.post("/admin/pay-success")
async def pay_success(order_no: str):
    for order in PAY_ORDER_LIST:
        if order["order_no"] == order_no and order["status"] == "待付款":
            order["status"] = "已付款已发货"
            # 充值流量
            key = order["bind_key"]
            add = order["init_flow"]
            if key in USER_BALANCE:
                USER_BALANCE[key] += add
            else:
                USER_BALANCE[key] = add
            save_balance(USER_BALANCE)
            save_orders(PAY_ORDER_LIST)
            return {"code": 200, "msg": "充值成功", "order": order}
    return {"code": 400, "msg": "订单不存在或已处理"}

@app.get("/admin/all-order")
async def get_all_order():
    return {"code": 200, "data": PAY_ORDER_LIST}

# 健康检查
@app.get("/")
async def health_check():
    return {
        "status": "running",
        "note": "JSON持久化版，重启数据不丢失",
        "balance_file": str(BALANCE_FILE),
        "order_file": str(ORDER_FILE)
    }
```
